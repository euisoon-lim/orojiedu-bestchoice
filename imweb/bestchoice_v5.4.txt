    h+='<div class="bcNotes"><span><i></i>선택한 상품 (로고를 누르면 해당 상품 기준으로 강조, BEST 로고는 선정 이유)</span><span>보험료 단위: 원 / 월납</span>'+(ns.length?'<span>기준: '+esc(ns.join(' · '))+'</span>':'')+'</div>';
    return h;
  }

  /* ── 선정 이유 ── */
  function reasonHtml(c){
    var d=c.detail||{}, feat=(d.feature||[]).slice(), age='';
    feat=feat.filter(function(t){ var m=t.match(/가입\s*연령\s*[–\-:：]?\s*(.+)$/); if(m&&!age){ age=m[1].trim(); return false;} return true; });
    var h='<div class="bcChips">'+(age?'<span>가입 연령 <b>'+esc(age)+'</b></span>':'')+'<span>분야 <b>'+esc(c.name)+'</b></span><span>비교 대상 '+c.products.slice(1).map(function(p){return logo(p.insurer);}).join('')+'</span></div>';
    h+='<div class="bcRs">';
    var n=1;
    function row(cls,title,body){ h+='<div class="bcR bcAni '+cls+'" style="--d:'+(0.1+n*0.08).toFixed(2)+'s"><div class="bcNo">'+pad(n++)+'</div><div class="bcT">'+title+'</div><div class="bcB">'+body+'</div></div>'; }
    if(feat.length) row('', '상품 특징', '<ul>'+feat.map(function(t){return '<li>'+esc(t)+'</li>';}).join('')+'</ul>');
    if(d.reason&&d.reason.length) row('bcKey', '선정 이유', '<ul>'+d.reason.map(function(t){return '<li>'+esc(t)+'</li>';}).join('')+'</ul>');
    if(d.sales&&d.sales.length) row('', '세일즈 포인트', d.sales.map(function(t){return '<p>'+esc(t)+'</p>';}).join(''));
    h+='</div>';
    return h;
  }

  /* ── 상세 ── */
  function renderDetail(){
    var s=sec(), c=cat(), isR=st.view==='reason', p=c.products[isR?0:st.prod]||c.products[0], ns=notes(c), h='';
    h+='<div class="bcTop"><div><a data-go="list">베스트 초이스</a><i>›</i><a data-go="list">'+esc(s.name)+'</a><i>›</i>'+(isR?'<a data-view="table">'+esc(c.name)+'</a><i>›</i><b>선정 이유</b>':'<b>'+esc(c.name)+'</b>')+'</div><button data-go="list">← 목록으로</button></div>';
    h+='<div class="bcDHead bcAni"><div><p class="bcEyebrow">'+esc(s.name)+' · '+esc(c.name)+(isR?' · 선정 이유':' · 타사 비교표')+'</p>'+'<span id="bcDLogo">'+logo(p.insurer)+'</span><h2 class="bcH1"><span class="bcHt" id="bcDTitle">'+esc(p.name)+'</span>'+(ns.length?'<small>'+esc(ns.join(' · '))+'</small>':'')+'</h2></div>'
      +'<div class="bcViews"><button data-view="table" class="'+(isR?'':'bcOn')+'">타사 비교표</button><button data-view="reason" class="'+(isR?'bcOn':'')+'">선정 이유</button></div></div>';
    h+='<div class="bcAni" style="--d:.12s">'+(isR ? reasonHtml(c) : tableHtml(c))+'</div>';
    h+='<div class="bcOther"><span>'+esc(s.name)+' 다른 분야</span>'; s.categories.forEach(function(x,i){ h+='<button data-cat="'+i+'" class="'+(i===st.cat?'bcOn':'')+'">'+esc(x.name)+'</button>'; }); h+='</div>';
    h+='<div class="bcDisc"><p>'+FOOT1+'</p><p>'+FOOT2+'</p></div>';
    h+='<div class="bcNav"><button data-go="list">← '+esc(s.name)+' 목록</button>'+(isR?'<button data-view="table">타사 비교표 보기</button>':'<button class="bcPri" data-view="reason">선정 이유 · 세일즈 포인트 →</button>')+'</div>';
    root.innerHTML=h;
  }

  function render(scroll){
    if(st.view==='list') renderList(); else renderDetail();
    if(scroll){ var y=root.getBoundingClientRect().top+window.pageYOffset-100; window.scrollTo({top:Math.max(0,y),behavior:'smooth'}); }
  }
  root.addEventListener('click',function(e){
    var b;
    if((b=e.target.closest('[data-sec]'))){ st.sec=+b.getAttribute('data-sec'); st.view='list'; render(false); return; }
    if((b=e.target.closest('[data-open]'))){ var a=b.getAttribute('data-open').split(':'); st.cat=+a[0]; st.prod=+a[1]; st.view='table'; render(true); return; }
    if((b=e.target.closest('[data-reason]'))){ st.view='reason'; render(true); return; }
    if((b=e.target.closest('[data-sel]'))){ st.prod=+b.getAttribute('data-sel'); applySel(); return; }
    if((b=e.target.closest('.bcTbl .bcCol'))){ /* v4.1: 열 아무 셀 클릭 → 해당 상품 선택 */
      var c=cat(); if(!c.table) return; var cr=colRanges(c.table), cs=+b.getAttribute('data-cs');
      for(var i=0;i<cr.ranges.length;i++){ if(cs>=cr.ranges[i].s && cs<=cr.ranges[i].e){ if(i!==st.prod){ st.prod=i; applySel(); } return; } }
      return;
    }
    if((b=e.target.closest('[data-view]'))){ st.view=b.getAttribute('data-view'); render(true); return; }
    if((b=e.target.closest('[data-cat]'))){ st.cat=+b.getAttribute('data-cat'); st.prod=0; if(st.view==='list') st.view='table'; render(true); return; }
    if((b=e.target.closest('[data-go]'))){ st.view='list'; st.prod=0; render(true); return; }
  });
  render(false);
}
