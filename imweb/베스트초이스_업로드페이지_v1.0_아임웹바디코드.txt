<script>
(function(){
  var FN='https://aztywbgtugactoumvmum.supabase.co/functions/v1/bestchoice-upload';
  var $=function(id){return document.getElementById(id);};
  var pw=$('bcUpPw'), drop=$('bcUpDrop'), fin=$('bcUpFile'), btn=$('bcUpBtn'), bar=$('bcUpBar'), steps=$('bcUpSteps'), msg=$('bcUpMsg');
  var file=null, busy=false, pollTimer=null;

  function api(b){ return fetch(FN,{method:'POST',headers:{'content-type':'application/json'},body:JSON.stringify(b)}).then(function(r){return r.json().then(function(j){ if(!r.ok) throw new Error(j.error||('HTTP '+r.status)); return j; });}); }
  function say(t,cls){ msg.textContent=t||''; msg.className='msg'+(cls?' '+cls:''); }
  function step(n,state){ var li=steps.querySelector('[data-s="'+n+'"]'); if(!li) return; li.className=state||''; }
  function fmt(b){ return b>1048576?(b/1048576).toFixed(1)+' MB':Math.round(b/1024)+' KB'; }
  function ko(iso){ try{ return new Date(iso).toLocaleString('ko-KR',{month:'numeric',day:'numeric',hour:'2-digit',minute:'2-digit'}); }catch(e){ return iso; } }
  function esc(s){ return String(s==null?'':s).replace(/[&<>"]/g,function(c){return{'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;'}[c];}); }

  /* 비밀번호 기억(이 브라우저 세션 동안) */
  try{ var saved=sessionStorage.getItem('bcUpPw'); if(saved){ pw.value=saved; } }catch(e){}
  function unlock(){ var ok=pw.value.trim().length>0; drop.classList.toggle('dis',!ok); btn.disabled=!(ok&&file&&!busy); }
  pw.addEventListener('input',unlock);
  pw.addEventListener('change',function(){ try{ sessionStorage.setItem('bcUpPw',pw.value); }catch(e){} refresh(); });

  /* 파일 선택 */
  function pick(f){
    if(!f) return;
    if(!/\.pptx$/i.test(f.name)){ say('PPTX 파일만 업로드할 수 있습니다.','err'); return; }
    file=f; $('bcUpName').textContent=f.name; $('bcUpSize').textContent=fmt(f.size); $('bcUpInfo').style.display='flex'; say(''); unlock();
  }
  drop.addEventListener('click',function(){ if(!drop.classList.contains('dis')) fin.click(); });
  fin.addEventListener('change',function(){ pick(fin.files[0]); });
  ['dragenter','dragover'].forEach(function(ev){ drop.addEventListener(ev,function(e){ e.preventDefault(); if(!drop.classList.contains('dis')) drop.classList.add('over'); }); });
  ['dragleave','drop'].forEach(function(ev){ drop.addEventListener(ev,function(e){ e.preventDefault(); drop.classList.remove('over'); }); });
  drop.addEventListener('drop',function(e){ if(drop.classList.contains('dis')) return; pick(e.dataTransfer.files[0]); });

  /* 업로드 → 알림 → 상태 폴링 */
  btn.addEventListener('click',function(){
    if(!file||busy) return; busy=true; unlock();
    steps.style.display='flex'; [1,2,3,4].forEach(function(n){step(n,'');}); step(1,'on'); bar.style.display='block'; bar.firstElementChild.style.width='5%'; say('업로드 URL 발급 중…');
    var pass=pw.value, signed, path;
    api({action:'sign',passcode:pass,filename:file.name}).then(function(s){
      signed=s; path=s.path; say('파일 업로드 중…');
      return new Promise(function(res,rej){
        var x=new XMLHttpRequest(); x.open('PUT',s.signedUrl,true);
        x.setRequestHeader('Content-Type','application/vnd.openxmlformats-officedocument.presentationml.presentation');
        x.setRequestHeader('x-upsert','false');
        x.upload.onprogress=function(e){ if(e.lengthComputable) bar.firstElementChild.style.width=(5+Math.round(e.loaded/e.total*55))+'%'; };
        x.onload=function(){ (x.status>=200&&x.status<300)?res():rej(new Error('스토리지 업로드 실패 ('+x.status+')')); };
        x.onerror=function(){ rej(new Error('네트워크 오류로 업로드하지 못했습니다.')); };
        x.send(file);
      });
    }).then(function(){
      step(1,'done'); step(2,'on'); bar.firstElementChild.style.width='65%'; say('변환 요청 전달 중…');
      return api({action:'notify',passcode:pass,path:path,filename:file.name,size:file.size});
    }).then(function(){
      step(2,'done'); step(3,'on'); bar.firstElementChild.style.width='75%'; say('GitHub에서 변환 중입니다. 보통 1~2분 걸려요. 이 화면을 열어두면 완료를 알려드립니다.');
      poll(Date.now());
    }).catch(function(e){
      var cur=steps.querySelector('li.on'); if(cur) cur.className='fail';
      say('실패: '+e.message,'err'); bar.style.display='none'; busy=false; unlock();
    });
  });

  var baseGen=null;
  function poll(startedAt){
    clearTimeout(pollTimer);
    api({action:'status',passcode:pw.value}).then(function(st){
      render(st);
      var run=(st.runs||[])[0];
      var fresh = st.latest && st.latest.generated_at && new Date(st.latest.generated_at).getTime() > startedAt-60000;
      if(run && run.status==='completed' && run.conclusion==='success' && fresh){
        step(3,'done'); step(4,'done'); bar.firstElementChild.style.width='100%';
        say('반영 완료 — '+(st.latest.month||'')+'호가 베스트 초이스 페이지에 적용되었습니다.','ok');
        busy=false; file=null; $('bcUpInfo').style.display='none'; fin.value=''; unlock(); return;
      }
      if(run && run.status==='completed' && run.conclusion!=='success' && new Date(run.created_at).getTime()>startedAt-60000){
        step(3,'fail'); say('변환 실패 — PPTX 구성을 확인해 주세요. 상세: ','err');
        var a=document.createElement('a'); a.href=run.url; a.target='_blank'; a.textContent='GitHub 실행 로그'; msg.appendChild(a);
        bar.style.display='none'; busy=false; unlock(); return;
      }
      if(Date.now()-startedAt > 6*60*1000){ say('아직 처리 중입니다. 잠시 후 이 페이지를 새로고침해 반영 상태를 확인해 주세요.'); busy=false; unlock(); return; }
      pollTimer=setTimeout(function(){ poll(startedAt); }, 8000);
    }).catch(function(){ pollTimer=setTimeout(function(){ poll(startedAt); }, 10000); });
  }

  /* 상태 표시 */
  function render(st){
    var L=st.latest, kv='';
    if(L){ kv+='<dt>반영 호</dt><dd><b>'+esc(L.month)+'</b></dd><dt>원본 파일</dt><dd>'+esc(L.source||'')+'</dd><dt>갱신 시각</dt><dd>'+esc(ko(L.generated_at))+'</dd>'; }
    else kv+='<dt>상태</dt><dd>반영 데이터를 읽지 못했습니다</dd>';
    $('bcUpLatest').innerHTML=kv;
    var h='';
    (st.uploads||[]).forEach(function(u){ h+='<li><span>'+esc(u.file_name)+(u.size_bytes?' <em>'+fmt(u.size_bytes)+'</em>':'')+'</span><span><em>'+esc(ko(u.uploaded_at))+'</em> <span class="tag '+(u.dispatched?'ok':'bad')+'">'+(u.dispatched?'전달됨':'전달 실패')+'</span></span></li>'; });
    (st.runs||[]).slice(0,3).forEach(function(r){ var c=r.status!=='completed'?'run':(r.conclusion==='success'?'ok':'bad'); var t=r.status!=='completed'?'변환 중':(r.conclusion==='success'?'변환 성공':'변환 실패'); h+='<li><span>변환 실행 · <a href="'+esc(r.url)+'" target="_blank" rel="noopener">로그</a></span><span><em>'+esc(ko(r.created_at))+'</em> <span class="tag '+c+'">'+t+'</span></span></li>'; });
    $('bcUpHist').innerHTML=h||'<li>이력이 없습니다</li>';
  }
  function refresh(){ if(!pw.value) return; api({action:'status',passcode:pw.value}).then(render).catch(function(e){ say(e.message,'err'); }); }
  unlock(); if(pw.value) refresh();
})();
</script>
